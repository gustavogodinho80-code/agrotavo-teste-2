function doPost(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet(); const dados = JSON.parse(e.postData.contents); if (dados.action === 'add') { let aba = sheet.getSheetByName(dados.tabela); if (!aba) { aba = sheet.insertSheet(dados.tabela); aba.appendRow(['id', 'data', 'dados']); } aba.appendRow([ dados.dados.id, new Date(), JSON.stringify(dados.dados) ]); return ContentService.createTextOutput(JSON.stringify({success: true})) .setMimeType(ContentService.MimeType.JSON); } if (dados.action === 'delete') { const aba = sheet.getSheetByName(dados.tabela); if (aba) { const values = aba.getDataRange().getValues(); for (let i = values.length - 1; i >= 1; i--) { if (values[i][0] == dados.id) { aba.deleteRow(i + 1); break; } } } return ContentService.createTextOutput(JSON.stringify({success: true})) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet(); const tabela = e.parameter.tabela; const aba = sheet.getSheetByName(tabela); if (!aba) { return ContentService.createTextOutput(JSON.stringify({data: []})) .setMimeType(ContentService.MimeType.JSON); } const values = aba.getDataRange().getValues(); const data = []; for (let i = 1; i < values.length; i++) { if (values[i][2]) { data.push(JSON.parse(values[i][2])); } } return ContentService.createTextOutput(JSON.stringify({data: data})) .setMimeType(ContentService.MimeType.JSON); }
